async function checkAppPassword() { const entered = document.getElementById('appPassword').value; const err = document.getElementById('errorMsg'); const warnBox = document.getElementById('securityWarningBox'); const metaText = document.getElementById('deviceMetaText'); const enteredHash = await sha256Hex(entered); if (enteredHash === PASSCODE_HASH) { localStorage.setItem('sk_app_unlocked', 'true'); document.getElementById('loginOverlay').style.display = 'none'; err.innerText = ""; if (warnBox) warnBox.classList.add('hidden'); } else { err.innerText = "Incorrect password! Please try again."; document.getElementById('appPassword').value = ""; if (warnBox) { warnBox.classList.remove('hidden'); const nowStr = new Date().toLocaleString('en-GB', { timeZone: 'Asia/Dubai' }); const platform = navigator.platform || 'Mobile Device'; if (metaText) { metaText.textContent = `Logged: ${nowStr} • Device: ${platform}`; } } } }
